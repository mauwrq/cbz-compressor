cmake_minimum_required(VERSION 3.10)
project(panelpress)
find_package(OpenCV REQUIRED)
find_package(Boost REQUIRED COMPONENTS program_options)
find_package(libzip REQUIRED)
include_directories(${OpenCV_INCLUDE_DIRS})
include_directories(include)
file(GLOB SOURCES "src/*.cpp")
list(REMOVE_ITEM SOURCES ${CMAKE_SOURCE_DIR}/src/main.cpp)
add_library(panelpress_lib STATIC ${SOURCES})
target_link_libraries(panelpress_lib PUBLIC ${OpenCV_LIBS} Boost::program_options libzip::zip)
add_executable(main src/main.cpp)
target_link_libraries(main panelpress_lib)
set_target_properties(main PROPERTIES OUTPUT_NAME cbz-compressor)
install(TARGETS main)

enable_testing()
find_package(GTest REQUIRED)
add_executable(tests tests/cleanup_tests.cpp)
target_link_libraries(tests panelpress_lib GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(tests DISCOVERY_MODE PRE_TEST)
